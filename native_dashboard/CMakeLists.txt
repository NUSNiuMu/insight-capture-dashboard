cmake_minimum_required(VERSION 3.16)
project(insight_native_dashboard LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTORCC ON)
find_package(Qt6 6.2 REQUIRED COMPONENTS Core Gui Quick QuickControls2 Quick3D Network WebSockets)
find_package(PkgConfig REQUIRED)
pkg_check_modules(GST REQUIRED IMPORTED_TARGET gstreamer-1.0 gstreamer-video-1.0 gstreamer-webrtc-1.0 gstreamer-sdp-1.0)
add_executable(insight-native main.cpp dashboard.cpp dashboard.h video.cpp video.h geometry.cpp geometry.h model.cpp model.h resources.qrc)
find_path(JETSON_MM_INCLUDE nvbufsurftransform.h PATHS "${CMAKE_BINARY_DIR}/jetson-headers" /usr/src/jetson_multimedia_api/include REQUIRED)
target_include_directories(insight-native PRIVATE ${JETSON_MM_INCLUDE})
target_compile_definitions(insight-native PRIVATE GST_USE_UNSTABLE_API)
target_link_libraries(insight-native PRIVATE Qt6::Quick Qt6::QuickControls2 Qt6::Quick3D Qt6::Network Qt6::WebSockets PkgConfig::GST)
install(TARGETS insight-native RUNTIME DESTINATION bin)
