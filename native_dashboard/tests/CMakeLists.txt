cmake_minimum_required(VERSION 3.16)
project(insight_task_actions_test LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTORCC ON)
find_package(Qt6 6.2 REQUIRED COMPONENTS Core Gui Quick QuickControls2 Quick3D Network WebSockets)
add_executable(task-actions task_actions.cpp ../dashboard.cpp ../dashboard.h ../workspace.cpp ../workspace.h ../geometry.cpp ../geometry.h ../model.cpp ../model.h ../resources.qrc)
target_include_directories(task-actions PRIVATE ..)
target_link_libraries(task-actions PRIVATE Qt6::Quick Qt6::QuickControls2 Qt6::Quick3D Qt6::Network Qt6::WebSockets)
enable_testing()
add_test(NAME task-actions COMMAND task-actions)
set_tests_properties(task-actions PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen;QT_QUICK_BACKEND=software;QT_QUICK_CONTROLS_STYLE=Basic")
